class AgentRouter(BaseModel):
    action: str = Field(description="Phân loại nhiệm vụ: 'voice', 'trend', 'code', 'mmo', 'image', 'auto', hoặc 'general'")
    data: str = Field(description="Dữ liệu, từ khóa, hoặc câu lệnh chi tiết đã được trích xuất để giao cho các AI con")

# ─── 3. CÁC TRƯỞNG PHÒNG AI (CHILD AGENTS) - CHỜ GẮN API KEY ──────────────────

class AgentVoice:
    """Trưởng phòng Audio: Sử dụng Edge-TTS (Đã hoàn thiện 100%)"""
    @staticmethod
    async def render(text: str, user_id: int, context: ContextTypes.DEFAULT_TYPE, chat_id: int):
        output_file = f"voice_{user_id}.mp3"
        status_msg = await context.bot.send_message(chat_id=chat_id, text="⏳ <i>[Agent Voice] Đang render âm thanh...</i>", parse_mode="HTML")
        try:
            communicate = edge_tts.Communicate(text, "vi-VN-NamMinhNeural")
            await communicate.save(output_file)
            with open(output_file, 'rb') as audio_file:
                await context.bot.send_audio(chat_id=chat_id, audio=audio_file)
            await status_msg.delete()
        except Exception as e:
            await status_msg.edit_text(f"❌ [Agent Voice] Thất bại: {e}")
        finally:
            if os.path.exists(output_file): os.remove(output_file)

class AgentData:
    """Trưởng phòng Data: Cào dữ liệu mạng (Đã hoàn thiện 100%)"""
    @staticmethod
    def search_trend(keyword: str) -> str:
        try:
            results = DDGS().text(keyword, max_results=5)
            if not results: return "❌ Không tìm thấy trend."
            resp = f"🌍 <b>DATA AGENT BÁO CÁO: '{keyword}'</b>\n\n"
            for idx, res in enumerate(results):
                resp += f"{idx+1}. <b>{res['title']}</b>\n- <i>{res['body']}</i>\n🔗 <a href='{res['href']}'>Link</a>\n\n"
            return resp
        except Exception as e:
            return f"❌ Lỗi cào data: {e}"

class AgentGemini:
    """Trưởng phòng Đa Năng & MMO: Xử lý văn bản bằng Gemini Pro/Flash"""
    @staticmethod
    def chat(system_prompt: str, user_text: str) -> str:
        if not gemini_client: return "❌ Lỗi: Chưa có Gemini API Key."
        try:
            resp = gemini_client.models.generate_content(
                model="gemini-2.0-flash", 
                config=types.GenerateContentConfig(system_instruction=system_prompt),
                contents=user_text,
            )
            return resp.text
        except Exception as e:
            return f"❌ <b>API KIỆT QUỆ:</b> {e}"

# ─── 4. SẾP TỔNG ĐIỀU PHỐI (THE ORCHESTRATOR) ──────────────────────────────────
async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    if await restrict_access(update): return
    text = update.message.text.strip()
    
    if text == "🛸 MỞ TRẠM ĐIỀU KHIỂN AI CENTRAL":
        await cmd_start(update, context)
        return
        
    await context.bot.send_chat_action(chat_id=update.effective_chat.id, action="typing")
    
    # 1. ÉP GEMINI TRẢ VỀ ĐÚNG SCHEMA CỦA AGENT ROUTER
    try:
        routing_instruction = """
        Bạn là Tổng Giám Đốc AI. Nhiệm vụ: Đọc lệnh của chủ nhân và phân công cho đúng Trưởng phòng.
        - Tạo giọng đọc/audio -> action: 'voice'
        - Tìm trend/tin tức -> action: 'trend'
        - Lập trình/Sửa code -> action: 'code'
        - Lên plan MMO/Kịch bản -> action: 'mmo'
        - Vẽ ảnh/Tạo hình ảnh -> action: 'image'
        - Đăng bài tự động/Nuôi nick -> action: 'auto'
        - Chào hỏi/Trò chuyện -> action: 'general'
        """
        response = gemini_client.models.generate_content(
            model="gemini-2.0-flash",
            config=types.GenerateContentConfig(
                system_instruction=routing_instruction,
                response_mime_type="application/json",
                response_schema=AgentRouter, # Tối thượng: Định chuẩn dữ liệu đầu ra 100%
            ),
            contents=text,
        )
        # Parse chuỗi JSON chuẩn mực từ Sếp Tổng
        route_plan = json.loads(response.text)
        action = route_plan.get("action")
        data = route_plan.get("data", "")
        
    except Exception as route_err:
        logger.error(f"Lỗi Sếp Tổng: {route_err}")
        await update.message.reply_text(f"❌ <b>SẾP TỔNG TẠM NGHỈ (429):</b> Hạn mức API đã cạn kiệt, vui lòng nâng cấp thẻ Pay-as-you-go!")
        return

    # 2. PHÂN LUỒNG XUỐNG CÁC TRƯỞNG PHÒNG AI (CHỈ ĐẠO THỰC THI)
    if action == "voice":
        await AgentVoice.render(data, update.effective_user.id, context, update.effective_chat.id)
        
    elif action == "trend":
        await update.message.reply_text(f"⏳ <i>[Agent Data] Đang quét tin tức: '{data}'...</i>", parse_mode="HTML")
        await update.message.reply_text(AgentData.search_trend(data), parse_mode="HTML", disable_web_page_preview=True)
        
    elif action == "code":
        await update.message.reply_text(f"👨‍💻 <b>[Agent Claude - Coder] Báo cáo:</b>\n\n{AgentClaude.generate_code(data)}", parse_mode="HTML")
        
    elif action == "mmo":
        mmo_prompt = "Bạn là Chuyên gia MMO & Automation. Đưa ra chiến lược chuyển đổi cao."
        await update.message.reply_text(f"💡 <b>[Agent Gemini - MMO] Báo cáo:</b>\n\n{AgentGemini.chat(mmo_prompt, data)}", parse_mode="HTML")
        
    elif action == "image":
        await update.message.reply_text(AgentMidjourney.generate_image(data), parse_mode="HTML")
        
    elif action == "auto":
        await update.message.reply_text(AgentPlaywright.run_automation(data), parse_mode="HTML")
        
    else:
        # Chat thông thường
        await update.message.reply_text(AgentGemini.chat("Bạn là Trợ lý AI trung tâm.", text))

    return ReplyKeyboardMarkup([[KeyboardButton("🛸 MỞ TRẠM ĐIỀU KHIỂN AI CENTRAL")]], resize_keyboard=True, is_persistent=True)

    return InlineKeyboardMarkup([
        [InlineKeyboardButton("👨‍💻 Kỹ Sư Claude (Code)", callback_data="none"), InlineKeyboardButton("🎨 Midjourney (Vẽ Ảnh)", callback_data="none")],
        [InlineKeyboardButton("📈 Data Agent (Trend)", callback_data="none"), InlineKeyboardButton("⚙️ Playwright (Auto)", callback_data="none")],
        [InlineKeyboardButton("🌐 Mở Kho Cấu Hình Hệ Thống", web_app=WebAppInfo(url=WEB_APP_URL))]
    ])

async def cmd_start(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    text = (
        "👑 <b>HỆ THỐNG ENTERPRISE MULTI-AGENT V9.0</b>\n\n"
        "Kiến trúc đã được nâng cấp lên mức Tối Cao. Bộ não Gemini hiện đóng vai trò Sếp Tổng, sở hữu khả năng kích hoạt các Class AI độc lập:\n\n"
        "✅ <b>Agent Audio:</b> Sẵn sàng\n"
        "✅ <b>Agent Data:</b> Sẵn sàng\n"
        "⏳ <b>Agent Claude:</b> Chờ nạp API\n"
        "⏳ <b>Agent Midjourney:</b> Chờ nạp API\n"
        "⏳ <b>Agent Playwright:</b> Chờ nạp Script\n\n"
        "<i>Chỉ cần ra lệnh tự nhiên, Sếp Tổng sẽ tự chia việc cho đúng bộ phận!</i>"
    )
    await update.message.reply_text(text, parse_mode="HTML", reply_markup=get_bottom_menu())
    await update.message.reply_text("🎛️ <b>SƠ ĐỒ TỔ CHỨC AI:</b>", parse_mode="HTML", reply_markup=get_inline_dashboard())

async def handle_callback(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.callback_query.answer("Module này hoạt động qua lệnh chat tự nhiên ngầm từ Sếp Tổng!")

def main() -> None:
    app = Application.builder().token(TELEGRAM_TOKEN).build()
    app.add_handler(CommandHandler("start", cmd_start))
    app.add_handler(CallbackQueryHandler(handle_callback))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))
    
    logger.info("🚀 Enterprise Multi-Agent V9.0 Kích Hoạt...")
    app.run_polling(allowed_updates=Update.ALL_TYPES)

if __name__ == "__main__":
    main()
